/* ========================================================================= מערכת סנכרון רב-משתמשים, אבטחת Firebase, וגיבוי חיצוני ========================================================================= */ const DB_LOCAL_KEY = 'hebrew_life_journal_v8'; let isRegisterMode = false; let cloudSyncUnsubscribe = null; // טעינת נתונים בסיסית מהמכשיר function getLocalJournalData() { const raw = localStorage.getItem(DB_LOCAL_KEY); if (!raw) { return { entries: {}, schedules: {}, dailyGoals: {}, dailyReflections: {}, weeklyBoxes: {}, yearlyGoals: [], parshiotCache: {} }; } try { return JSON.parse(raw); } catch(e) { return {}; } } function saveLocalJournalData(data) { localStorage.setItem(DB_LOCAL_KEY, JSON.stringify(data)); } // -------------------------------------------------------------------------- // 1. ניהול הזדהות ומשתמשים (Auth & Forgot Password) // -------------------------------------------------------------------------- function openAuthModal() { document.getElementById('authModalOverlay').style.display = 'flex'; document.getElementById('authErrorMsg').textContent = ''; } function closeAuthModal() { document.getElementById('authModalOverlay').style.display = 'none'; } function toggleAuthMode() { isRegisterMode = !isRegisterMode; document.getElementById('authModalTitle').textContent = isRegisterMode ? 'הרשמת משתמש חדש' : 'התחברות למערכת'; document.getElementById('btnSubmitAuth').textContent = isRegisterMode ? 'הירשם והתחבר' : 'התחבר'; document.getElementById('authToggleLink').textContent = isRegisterMode ? 'כבר רשום? התחבר כאן' : 'אין לך חשבון? הירשם כאן'; document.getElementById('authForgotLink').style.display = isRegisterMode ? 'none' : 'block'; document.getElementById('authErrorMsg').textContent = ''; } // התחברות מהירה באמצעות Google async function loginWithGoogle() { const provider = new firebase.auth.GoogleAuthProvider(); try { await firebase.auth().signInWithPopup(provider); closeAuthModal(); } catch (err) { document.getElementById('authErrorMsg').textContent = err.message; } } // התחברות / הרשמה עם אימייל וסיסמה async function handleEmailPasswordAuth() { const email = document.getElementById('authEmail').value.trim(); const password = document.getElementById('authPassword').value; const errEl = document.getElementById('authErrorMsg'); errEl.textContent = ''; if (!email || !password) return errEl.textContent = 'יש להזין אימייל וסיסמה.'; if (password.length < 6) return errEl.textContent = 'הסיסמה חייבת להכיל לפחות 6 תווים.'; try { if (isRegisterMode) { await firebase.auth().createUserWithEmailAndPassword(email, password); } else { await firebase.auth().signInWithEmailAndPassword(email, password); } closeAuthModal(); } catch (err) { if (err.code === 'auth/user-not-found' || err.code === 'auth/wrong-password') { errEl.textContent = 'אימייל או סיסמה שגויים.'; } else if (err.code === 'auth/email-already-in-use') { errEl.textContent = 'כתובת האימייל הזו כבר רשומה במערכת.'; } else { errEl.textContent = err.message; } } } // שחזור סיסמה במייל (Forgot Password) async function handleForgotPassword() { const email = document.getElementById('authEmail').value.trim() || prompt("הזן את כתובת האימייל שאיתה נרשמת:"); if (!email) return; try { await firebase.auth().sendPasswordResetEmail(email); alert(`קישור לאיפוס הסיסמה נשלח לכתובת:\n${email}\nבדוק את תיבת הדואר הנכנס (והספאם).`); } catch (err) { alert("שגיאה בשליחת קישור איפוס: " + err.message); } } // התנתקות async function handleSignOut() { if (cloudSyncUnsubscribe) cloudSyncUnsubscribe(); await firebase.auth().signOut(); location.reload(); } // -------------------------------------------------------------------------- // 2. סנכרון ענן בזמן אמת בין מכשירים (מחשב <-> טלפון) // -------------------------------------------------------------------------- firebase.auth().onAuthStateChanged(user => { const loginBtn = document.getElementById('btnLoginOpen'); const userBadge = document.getElementById('userProfileBadge'); const displayName = document.getElementById('userDisplayName'); const syncStatus = document.getElementById('cloudSyncStatus'); if (user) { loginBtn.style.display = 'none'; userBadge.style.display = 'flex'; displayName.textContent = user.displayName || user.email.split('@')[0]; syncStatus.textContent = '☁️ מסונכרן לענן'; syncStatus.style.color = '#4ade80'; startRealtimeCloudSync(user.uid); } else { loginBtn.style.display = 'block'; userBadge.style.display = 'none'; syncStatus.textContent = 'מצב מקומי (לא מסונכרן)'; syncStatus.style.color = '#94a3b8'; if (cloudSyncUnsubscribe) cloudSyncUnsubscribe(); } }); // האזנה לסנכרון בזמן אמת מה-Firestore של המשתמש function startRealtimeCloudSync(uid) { const userDocRef = firebase.firestore().collection('users').doc(uid); cloudSyncUnsubscribe = userDocRef.onSnapshot(doc => { if (doc.exists) { const cloudData = doc.data().journalData; if (cloudData && typeof cloudData === 'object') { const localData = getLocalJournalData(); // בדיקה האם יש הבדל בין הענן למכשיר הנוכחי if (JSON.stringify(cloudData) !== JSON.stringify(localData)) { saveLocalJournalData(cloudData); if (typeof renderCurrentView === 'function') renderCurrentView(); console.log("הנתונים סונכרנו בהצלחה מהענן ממכשיר אחר!"); } } } else { // שמירה ראשונית אם המשתמש חדש לגמרי syncCurrentDataToCloud(); } }, err => console.warn("שגיאה בסנכרון ענן:", err)); } // פונקציה הנקראת בכל פעם שהמשתמש שומר רשומה או מעדכן לו"ז function syncCurrentDataToCloud() { const user = firebase.auth().currentUser; if (!user) return; // שמירה מקומית בלבד עד שמתחברים const currentData = getLocalJournalData(); firebase.firestore().collection('users').doc(user.uid).set({ journalData: currentData, lastUpdated: firebase.firestore.FieldValue.serverTimestamp(), userEmail: user.email }, { merge: true }).catch(e => console.error("שגיאה בהעלאה לענן:", e)); } // -------------------------------------------------------------------------- // 3. כלי גיבוי ושחזור (Docs / קובץ / שחזור מהיר) // -------------------------------------------------------------------------- // העתקת קוד להדבקה ב-Google Docs function copyBackupToClipboard() { const data = localStorage.getItem(DB_LOCAL_KEY) || '{}'; navigator.clipboard.writeText(data).then(() => { alert("קוד הגיבוי הועתק ללוח!\nפתח את מסמך ה-Google Docs שלך והדבק אותו שם (Ctrl + V)."); }).catch(() => { prompt("העתק את קוד הגיבוי להדבקה ב-Docs:", data); }); } // הורדת קובץ גיבוי למחשב או לטלפון function downloadBackupFile() { const data = localStorage.getItem(DB_LOCAL_KEY) || '{}'; const blob = new Blob([data], { type: 'application/json' }); const a = document.createElement('a'); a.href = URL.createObjectURL(blob); a.download = `יומן_אישי_גיבוי_${new Date().toISOString().split('T')[0]}.json`; a.click(); } // שחזור מקוד או הדבקה function openImportModal() { const code = prompt("הדבק כאן את קוד הגיבוי (מתוך מסמך ה-Google Docs או קובץ הגיבוי):"); if (!code) return; try { const parsed = JSON.parse(code.trim()); if (parsed && typeof parsed === 'object') { saveLocalJournalData(parsed); syncCurrentDataToCloud(); // עדכון מיידי בענן עבור כל שאר המכשירים alert("היומן שוחזר בהצלחה בכל המכשירים!"); location.reload(); } else { throw new Error(); } } catch(e) { alert("קוד הגיבוי שהודבק אינו תקין. ודא שהעתקת את כל הטקסט במלואו."); } }
